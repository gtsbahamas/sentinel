# Sentinel AI Layer Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a pluggable, opt-in AI analysis layer to sentinel that runs per-endpoint semantic security analysis via Anthropic or OpenAI, produces findings through the existing `Finding[]` pipeline, and ships safely (hard budget caps, graceful degradation, mock-first tests).

**Architecture:** Thin orchestrator + pluggable provider adapters. `runAI(model, config)` runs after `runRules` (if `--ai` flag set), iterates over `model.endpoints[]` with a bounded worker pool, calls a `AIProvider.generateFindings()` per endpoint with structured-output enforcement (forced tool-use on Anthropic, `json_schema strict` on OpenAI), validates output through Zod, and returns new findings tagged `layer: "ai"`. Optional `--ai-deep` runs a Stage A project-wide risk summary that biases endpoint priority order (advisory only, never frames Stage B findings). Prompt caching carries the project summary across endpoint calls.

**Tech Stack:** TypeScript, Node 20+, Vitest, Zod, Commander, `@anthropic-ai/sdk`, `openai`. Extends existing sentinel pipeline (`src/layers/recon`, `src/layers/rules`, `src/commands/scan.ts`).

```ts
import { z } from "zod";
import type { Finding } from "../../model/types.js";

export const AIFindingSchema = z.object({
  ruleId: z.string().regex(/^AI-[A-Z]+-\d{3}$/),
  domain: z.enum([
    "secrets-exposure",
    "input-injection",
    "auth-access-control",
    "file-upload",
    "api-security",
    "cryptography",
    "supply-chain",
    "infrastructure",
    "business-logic",
    "logging-monitoring",
  ]),
  severity: z.enum(["critical", "high", "medium", "low", "info"]),
  confidence: z.enum(["high", "medium", "low"]),
  title: z.string().min(10).max(200),
  description: z.string().min(20).max(2000),
  evidence: z
    .array(
      z.object({
        file: z.string(),
        line: z.number().int().positive().optional(),
        snippet: z.string().max(500).optional(),
      }),
    )
    .min(1),
  remediation: z.string().min(20).max(1000),
  reasoning: z.string().min(20).max(2000),
});

export type AIFinding = z.infer<typeof AIFindingSchema>;

export const StageARiskSchema = z.object({
  projectRisks: z.array(
    z.object({
      domain: AIFindingSchema.shape.domain,
      description: z.string().min(10),
      affectedEndpoints: z.array(z.string()),
      severity: AIFindingSchema.shape.severity,
    }),
  ),
  priorityOrder: z.array(z.string()),
});

export function aiToFinding(ai: AIFinding): Finding {
  const first = ai.evidence[0];
  return {
    id: `ai:${ai.ruleId}:${first.file}:${first.line ?? 0}`,
    domain: ai.domain,
    severity: ai.severity,
    confidence: ai.confidence,
    title: ai.title,
    description: ai.description,
    evidence: ai.evidence.map((e) => ({
      file: e.file,
      line: e.line,
      snippet: e.snippet,
      context: { reasoning: ai.reasoning },
    })),
    remediation: ai.remediation,
    references: [],
    layer: "ai",
    ruleId: ai.ruleId,
  };
}
```

```ts
import type { AIUsage } from "../../model/types.js";

export interface ProviderPricing {
  inputPerMTokUsd: number;
  outputPerMTokUsd: number;
  cacheReadPerMTokUsd: number;
  cacheWritePerMTokUsd: number;
}

export interface TokenEstimate {
  inputTokens: number;
  estimatedCostUsd: number;
}

export class BudgetTracker {
  private totalInput = 0;
  private totalOutput = 0;
  private totalCacheRead = 0;
  private totalCacheWrite = 0;

  constructor(
    private readonly maxCostUsd: number,
    private readonly pricing: ProviderPricing,
  ) {}

  recordUsage(usage: AIUsage): void {
    this.totalInput += usage.inputTokens;
    this.totalOutput += usage.outputTokens;
    this.totalCacheRead += usage.cacheReadTokens;
    this.totalCacheWrite += usage.cacheWriteTokens;
  }

  currentCostUsd(): number {
    const perM = (t: number, rate: number) => (t / 1_000_000) * rate;
    return (
      perM(this.totalInput, this.pricing.inputPerMTokUsd) +
      perM(this.totalOutput, this.pricing.outputPerMTokUsd) +
      perM(this.totalCacheRead, this.pricing.cacheReadPerMTokUsd) +
      perM(this.totalCacheWrite, this.pricing.cacheWritePerMTokUsd)
    );
  }

  wouldExceed(estimate: TokenEstimate): boolean {
    return this.currentCostUsd() + estimate.estimatedCostUsd > this.maxCostUsd;
  }

  exhausted(): boolean {
    return this.currentCostUsd() >= this.maxCostUsd;
  }

  summary(): AIUsage {
    return {
      inputTokens: this.totalInput,
      outputTokens: this.totalOutput,
      cacheReadTokens: this.totalCacheRead,
      cacheWriteTokens: this.totalCacheWrite,
      estimatedCostUsd: this.currentCostUsd(),
    };
  }
}
```

```ts
import type { TokenEstimate } from "../budget.js";

export interface MockDispatch {
  [userPromptKeyOrSubstring: string]: GenerateResponse;
}

/**
 * Deterministic test provider. Fails loudly on unknown keys and
 * on any attempt to use real network — tests rely on this.
 */
export class MockProvider implements AIProvider {
  readonly name = "mock" as const;
  public callCount = 0;

  constructor(private readonly dispatch: MockDispatch) {}

  async generateFindings(request: GenerateRequest): Promise<GenerateResponse> {
    this.callCount++;
```

```ts
import { readFile } from "fs/promises";
import { dirname, resolve as resolvePath, relative } from "path";
import type { ApiEndpoint, ProjectModel } from "../../model/types.js";
import type { AdjacentFile } from "./prompts.js";

export interface EndpointContext {
  handlerSource: string;
  adjacent: AdjacentFile[];
}

const CHARS_PER_TOKEN = 4;
const MAX_ADJACENT = 5;
const MAX_IMPORT_DEPTH = 2;

function approxTokens(s: string): number {
  return Math.ceil(s.length / CHARS_PER_TOKEN);
}

const IMPORT_RE = /import\s+(?:[\w{}*,\s]+from\s+)?["']([^"']+)["']/g;

function extractLocalImports(source: string): string[] {
  const results: string[] = [];
  let m: RegExpExecArray | null;
  IMPORT_RE.lastIndex = 0;
  while ((m = IMPORT_RE.exec(source)) !== null) {
    const spec = m[1];
    if (spec.startsWith(".") || spec.startsWith("/")) results.push(spec);
  }
  return results;
}

async function resolveImport(fromFile: string, spec: string, projectRoot: string): Promise<string | null> {
  const base = resolvePath(dirname(fromFile), spec);
  const candidates = [
    base,
    base + ".ts",
    base + ".tsx",
    base + ".js",
    base + ".jsx",
    resolvePath(base, "index.ts"),
    resolvePath(base, "index.tsx"),
  ].map((c) => c.replace(/\.js$/, ".ts"));
  for (const c of candidates) {
    try {
      await readFile(c, "utf-8");
      const rel = relative(projectRoot, c);
      if (rel.startsWith("..")) return null;
      return c;
    } catch {
      continue;
    }
  }
  return null;
}

export async function buildContext(
  endpoint: ApiEndpoint,
  model: ProjectModel,
  maxTokens: number,
): Promise<EndpointContext | null> {
  const projectRoot = model.meta.path;

  let handlerSource: string;
  try {
    handlerSource = await readFile(endpoint.file, "utf-8");
  } catch {
    return null;
  }

  if (approxTokens(handlerSource) > maxTokens) {
    return null; // handler alone exceeds cap
  }

  // Walk imports BFS up to MAX_IMPORT_DEPTH, collect up to MAX_ADJACENT files.
  const collected: AdjacentFile[] = [];
  const seen = new Set<string>([endpoint.file]);
  const queue: Array<{ file: string; depth: number }> = [{ file: endpoint.file, depth: 0 }];

  while (queue.length > 0 && collected.length < MAX_ADJACENT) {
    const { file, depth } = queue.shift()!;
    if (depth >= MAX_IMPORT_DEPTH) continue;
    let source: string;
    try {
      source = await readFile(file, "utf-8");
    } catch {
      continue;
    }
    for (const spec of extractLocalImports(source)) {
      const resolved = await resolveImport(file, spec, projectRoot);
      if (!resolved || seen.has(resolved)) continue;
      seen.add(resolved);
      let adjSource: string;
      try {
        adjSource = await readFile(resolved, "utf-8");
      } catch {
        continue;
      }
      collected.push({ path: relative(projectRoot, resolved), content: adjSource });
      if (collected.length >= MAX_ADJACENT) break;
      queue.push({ file: resolved, depth: depth + 1 });
    }
  }

  // Drop adjacent files in reverse until we fit under maxTokens.
  let adjacent = collected;
  let totalTokens =
    approxTokens(handlerSource) + adjacent.reduce((a, f) => a + approxTokens(f.content), 0);
  while (totalTokens > maxTokens && adjacent.length > 0) {
    adjacent = adjacent.slice(0, -1);
    totalTokens = approxTokens(handlerSource) + adjacent.reduce((a, f) => a + approxTokens(f.content), 0);
  }

  return { handlerSource, adjacent };
}
```

```ts
import type { AIConfig, AIError, AIRunResult, ApiEndpoint, Finding, ProjectModel } from "../../model/types.js";
import type { AIProvider, GenerateRequest } from "./providers/base.js";
import type { BudgetTracker } from "./budget.js";
import { buildContext } from "./context-builder.js";
import { buildProjectSummary, buildEndpointPrompt, SYSTEM_PREAMBLE, REPORT_FINDINGS_TOOL } from "./prompts.js";
import { AIFindingSchema, aiToFinding } from "./schema.js";

export async function runOrchestrator(
  model: ProjectModel,
  config: AIConfig,
  provider: AIProvider,
  budget: BudgetTracker,
  priorityOrder?: string[],
): Promise<AIRunResult> {
  const start = Date.now();
  const projectSummary = buildProjectSummary(model);
  const findings: Finding[] = [];
  const errors: AIError[] = [];
  let analyzed = 0;
  let skipped = 0;
  let budgetExhausted = false;

  const ordered = orderEndpoints(model.endpoints, priorityOrder);

  const concurrency = Math.max(1, config.concurrency);
  const queue = [...ordered];
  let halted = false;

  async function worker(): Promise<void> {
    while (!halted) {
      const endpoint = queue.shift();
      if (!endpoint) return;

      if (budget.exhausted()) {
        halted = true;
        budgetExhausted = true;
        skipped += queue.length + 1;
        return;
      }

      const ctx = await buildContext(endpoint, model, config.maxTokensPerEndpoint);
      if (!ctx) {
        skipped++;
        errors.push({ kind: "context", message: "handler exceeds per-endpoint token cap or unreadable", endpoint: endpoint.path });
        continue;
      }

      const userPrompt = buildEndpointPrompt(endpoint, ctx.handlerSource, ctx.adjacent);
      const request: GenerateRequest = {
        systemPrompt: SYSTEM_PREAMBLE,
        projectSummary,
        userPrompt,
        toolDefinition: REPORT_FINDINGS_TOOL,
        maxInputTokens: config.maxTokensPerEndpoint,
        maxOutputTokens: 2000,
      };

      const estimate = provider.estimateTokens(request);
      if (budget.wouldExceed(estimate)) {
        halted = true;
        budgetExhausted = true;
        skipped += queue.length + 1;
        return;
      }

      try {
        const response = await provider.generateFindings(request);
        budget.recordUsage(response.usage);

        if (response.stopReason === "error") {
          skipped++;
          errors.push({ kind: "provider", message: response.rawError ?? "provider error", endpoint: endpoint.path });
          continue;
        }

        analyzed++;
        for (const raw of response.rawFindings) {
          const parsed = AIFindingSchema.safeParse(raw);
          if (!parsed.success) {
            errors.push({ kind: "parse", message: parsed.error.message.slice(0, 300), endpoint: endpoint.path });
            continue;
          }
          findings.push(aiToFinding(parsed.data));
        }
      } catch (err) {
        skipped++;
        errors.push({
          kind: "provider",
          message: err instanceof Error ? err.message : String(err),
          endpoint: endpoint.path,
        });
      }
    }
  }

  const workers = Array.from({ length: concurrency }, () => worker());
  await Promise.all(workers);

  return {
    findings,
    usage: budget.summary(),
    stageA: null,
    errors,
    budgetExhausted,
    endpointsAnalyzed: analyzed,
    endpointsSkipped: skipped,
    durationMs: Date.now() - start,
  };
}

function orderEndpoints(endpoints: ApiEndpoint[], priorityOrder?: string[]): ApiEndpoint[] {
  if (!priorityOrder || priorityOrder.length === 0) return [...endpoints];
  const byPath = new Map(endpoints.map((e) => [e.path, e]));
  const ordered: ApiEndpoint[] = [];
  const seen = new Set<string>();
  for (const p of priorityOrder) {
    const e = byPath.get(p);
    if (e && !seen.has(p)) {
      ordered.push(e);
      seen.add(p);
    }
  }
  for (const e of endpoints) {
    if (!seen.has(e.path)) ordered.push(e);
  }
  return ordered;
}
```

```ts
import type { AIConfig, ProjectModel } from "../../model/types.js";
import type { AIProvider } from "./providers/base.js";
import { buildContext } from "./context-builder.js";
import { buildProjectSummary, buildEndpointPrompt, SYSTEM_PREAMBLE, REPORT_FINDINGS_TOOL } from "./prompts.js";

export interface DryRunResult {
  endpointCount: number;
  totalInputTokens: number;
  avgPerEndpointTokens: number;
  projectSummaryTokens: number;
  estimatedCostUsd: number;
  maxCostUsd: number;
  withinBudget: boolean;
}

export async function runDryRun(
  model: ProjectModel,
  config: AIConfig,
  provider: AIProvider,
): Promise<DryRunResult> {
  const summary = buildProjectSummary(model);
  const summaryEst = provider.estimateTokens({
    systemPrompt: SYSTEM_PREAMBLE,
    projectSummary: summary,
    userPrompt: "",
    toolDefinition: REPORT_FINDINGS_TOOL,
    maxInputTokens: config.maxTokensPerEndpoint,
    maxOutputTokens: 2000,
  });

  let totalInput = 0;
  let totalCost = 0;

  for (const endpoint of model.endpoints) {
    const ctx = await buildContext(endpoint, model, config.maxTokensPerEndpoint);
    if (!ctx) continue;
    const prompt = buildEndpointPrompt(endpoint, ctx.handlerSource, ctx.adjacent);
    const est = provider.estimateTokens({
      systemPrompt: SYSTEM_PREAMBLE,
      projectSummary: summary,
      userPrompt: prompt,
      toolDefinition: REPORT_FINDINGS_TOOL,
      maxInputTokens: config.maxTokensPerEndpoint,
      maxOutputTokens: 2000,
    });
    totalInput += est.inputTokens;
    totalCost += est.estimatedCostUsd;
  }

  return {
    endpointCount: model.endpoints.length,
    totalInputTokens: totalInput,
    avgPerEndpointTokens: model.endpoints.length > 0 ? Math.round(totalInput / model.endpoints.length) : 0,
    projectSummaryTokens: summaryEst.inputTokens,
    estimatedCostUsd: totalCost,
    maxCostUsd: config.maxCostUsd,
    withinBudget: totalCost <= config.maxCostUsd,
  };
}
```

```ts
import type { AIProvider, GenerateRequest, GenerateResponse } from "./base.js";
import type { ProviderPricing, TokenEstimate } from "../budget.js";

// Pricing per 1M tokens. Update these when Anthropic changes rates.
// Cache write is ~125% of input, cache read is ~10% of input (Anthropic policy 2026).
export const ANTHROPIC_PRICING: Record<string, ProviderPricing> = {
  "claude-sonnet-4-6": { inputPerMTokUsd: 3, outputPerMTokUsd: 15, cacheReadPerMTokUsd: 0.3, cacheWritePerMTokUsd: 3.75 },
  "claude-haiku-4-5": { inputPerMTokUsd: 1, outputPerMTokUsd: 5, cacheReadPerMTokUsd: 0.1, cacheWritePerMTokUsd: 1.25 },
};

interface MessagesCreateArgs {
  model: string;
  max_tokens: number;
  system: Array<{ type: "text"; text: string; cache_control?: { type: "ephemeral" } }>;
  messages: Array<{ role: "user"; content: string }>;
  tools: Array<{ name: string; description: string; input_schema: object }>;
  tool_choice: { type: "tool"; name: string };
}

interface MessagesCreateResponse {
  content: Array<{ type: string; name?: string; input?: unknown }>;
  stop_reason: "end_turn" | "tool_use" | "max_tokens" | string;
  usage: {
    input_tokens: number;
    output_tokens: number;
    cache_creation_input_tokens?: number;
    cache_read_input_tokens?: number;
  };
}

interface AnthropicLike {
  messages: { create: (args: MessagesCreateArgs) => Promise<MessagesCreateResponse> };
}

const CHARS_PER_TOKEN = 4;

export class AnthropicProvider implements AIProvider {
  readonly name = "anthropic" as const;
  private readonly pricing: ProviderPricing;

  constructor(
    private readonly model: string,
    private readonly client: AnthropicLike,
  ) {
    this.pricing = ANTHROPIC_PRICING[model] ?? ANTHROPIC_PRICING["claude-sonnet-4-6"];
  }

  async generateFindings(request: GenerateRequest): Promise<GenerateResponse> {
    try {
      const response = await this.client.messages.create({
        model: this.model,
        max_tokens: request.maxOutputTokens,
        system: [
          { type: "text", text: request.systemPrompt },
          { type: "text", text: request.projectSummary, cache_control: { type: "ephemeral" } },
        ],
        messages: [{ role: "user", content: request.userPrompt }],
        tools: [
          {
            name: request.toolDefinition.name,
            description: request.toolDefinition.description,
            input_schema: request.toolDefinition.schema,
          },
        ],
        tool_choice: { type: "tool", name: request.toolDefinition.name },
      });

      const inputTokens = response.usage.input_tokens;
      const outputTokens = response.usage.output_tokens;
      const cacheWrite = response.usage.cache_creation_input_tokens ?? 0;
      const cacheRead = response.usage.cache_read_input_tokens ?? 0;

      const rawFindings: unknown[] = [];
      for (const block of response.content) {
        if (block.type === "tool_use" && block.name === request.toolDefinition.name && block.input) {
          const input = block.input as { findings?: unknown[] };
          if (Array.isArray(input.findings)) rawFindings.push(...input.findings);
        }
      }

      const stopReason =
        response.stop_reason === "tool_use" || response.stop_reason === "end_turn" || response.stop_reason === "max_tokens"
          ? response.stop_reason
          : "error";

      return {
        rawFindings,
        stopReason,
        usage: {
          inputTokens,
          outputTokens,
          cacheReadTokens: cacheRead,
          cacheWriteTokens: cacheWrite,
          estimatedCostUsd: this.computeCost(inputTokens, outputTokens, cacheRead, cacheWrite),
        },
      };
    } catch (err) {
      return {
        rawFindings: [],
        stopReason: "error",
        usage: { inputTokens: 0, outputTokens: 0, cacheReadTokens: 0, cacheWriteTokens: 0, estimatedCostUsd: 0 },
        rawError: err instanceof Error ? err.message : String(err),
      };
    }
  }

  estimateTokens(request: GenerateRequest): TokenEstimate {
    const inputTokens = Math.ceil(
      (request.systemPrompt.length + request.projectSummary.length + request.userPrompt.length) / CHARS_PER_TOKEN,
    );
    const estimatedCostUsd = (inputTokens / 1_000_000) * this.pricing.inputPerMTokUsd;
    return { inputTokens, estimatedCostUsd };
  }

  private computeCost(input: number, output: number, cacheRead: number, cacheWrite: number): number {
    const perM = (t: number, r: number) => (t / 1_000_000) * r;
    return (
      perM(input - cacheRead - cacheWrite, this.pricing.inputPerMTokUsd) +
      perM(output, this.pricing.outputPerMTokUsd) +
      perM(cacheRead, this.pricing.cacheReadPerMTokUsd) +
      perM(cacheWrite, this.pricing.cacheWritePerMTokUsd)
    );
  }
}
```

```ts
export const OPENAI_PRICING: Record<string, ProviderPricing> = {
  "gpt-5": { inputPerMTokUsd: 5, outputPerMTokUsd: 20, cacheReadPerMTokUsd: 2.5, cacheWritePerMTokUsd: 5 },

  chat: {
    completions: {
      create: (args: ChatCompletionArgs) => Promise<ChatCompletionResponse>;
    };
  };
}

interface ChatCompletionArgs {
  model: string;
  messages: Array<{ role: "system" | "user"; content: string }>;
  response_format: {
    type: "json_schema";
    json_schema: { name: string; strict: true; schema: object };
  };
  max_completion_tokens: number;
}

interface ChatCompletionResponse {
  choices: Array<{
    message: { content: string | null };
    finish_reason: string;
  }>;
  usage: {
    prompt_tokens: number;
    completion_tokens: number;
    prompt_tokens_details?: { cached_tokens?: number };
  };
}

const CHARS_PER_TOKEN = 4;

export class OpenAIProvider implements AIProvider {
  readonly name = "openai" as const;
  private readonly pricing: ProviderPricing;

  constructor(
    private readonly model: string,
    private readonly client: OpenAILike,
  ) {
    this.pricing = OPENAI_PRICING[model] ?? OPENAI_PRICING["gpt-5"];
  }

  async generateFindings(request: GenerateRequest): Promise<GenerateResponse> {
    try {
      const response = await this.client.chat.completions.create({
        model: this.model,
        messages: [
          { role: "system", content: `${request.systemPrompt}\n\n${request.projectSummary}` },
          { role: "user", content: request.userPrompt },
        ],
        response_format: {
          type: "json_schema",
          json_schema: {
            name: request.toolDefinition.name,
            strict: true,
            schema: request.toolDefinition.schema,
          },
        },
        max_completion_tokens: request.maxOutputTokens,
      });

      const choice = response.choices[0];
      const content = choice?.message?.content ?? "";
      let rawFindings: unknown[] = [];
      let stopReason: GenerateResponse["stopReason"] = "end_turn";

      try {
        const parsed = JSON.parse(content) as { findings?: unknown[] };
        if (Array.isArray(parsed.findings)) rawFindings = parsed.findings;
      } catch {
        stopReason = "error";
      }

      const cached = response.usage.prompt_tokens_details?.cached_tokens ?? 0;
      return {
        rawFindings,
        stopReason,
        usage: {
          inputTokens: response.usage.prompt_tokens,
          outputTokens: response.usage.completion_tokens,
          cacheReadTokens: cached,
          cacheWriteTokens: 0,
          estimatedCostUsd: this.computeCost(response.usage.prompt_tokens, response.usage.completion_tokens, cached),
        },
        rawError: stopReason === "error" ? "malformed json content" : undefined,
      };
    } catch (err) {
      return {
        rawFindings: [],
        stopReason: "error",
        usage: { inputTokens: 0, outputTokens: 0, cacheReadTokens: 0, cacheWriteTokens: 0, estimatedCostUsd: 0 },
        rawError: err instanceof Error ? err.message : String(err),
      };
    }
  }

  estimateTokens(request: GenerateRequest): TokenEstimate {
    const inputTokens = Math.ceil(
      (request.systemPrompt.length + request.projectSummary.length + request.userPrompt.length) / CHARS_PER_TOKEN,
    );
    const estimatedCostUsd = (inputTokens / 1_000_000) * this.pricing.inputPerMTokUsd;
    return { inputTokens, estimatedCostUsd };
  }

  private computeCost(input: number, output: number, cached: number): number {
    const perM = (t: number, r: number) => (t / 1_000_000) * r;
    const freshInput = Math.max(0, input - cached);
    return (
      perM(freshInput, this.pricing.inputPerMTokUsd) +
      perM(cached, this.pricing.cacheReadPerMTokUsd) +
      perM(output, this.pricing.outputPerMTokUsd)
    );
  }
}
```

```ts
import type { AIProvider } from "./providers/base.js";

import { AnthropicProvider, ANTHROPIC_PRICING } from "./providers/anthropic.js";
import { OpenAIProvider, OPENAI_PRICING } from "./providers/openai.js";

import { runDryRun, type DryRunResult } from "./dry-run.js";

export { runDryRun };
export type { DryRunResult };

export async function runAI(
  model: ProjectModel,
  config: AIConfig,
  injectedProvider?: AIProvider,
): Promise<AIRunResult> {
  const provider = injectedProvider ?? createProvider(config);
  const pricing = pricingFor(config);
  const budget = new BudgetTracker(config.maxCostUsd, pricing);

  let priorityOrder: string[] | undefined;
  let stageA = null;
  if (config.deep) {
    stageA = await runStageA(model, provider, budget);
    priorityOrder = stageA?.priorityOrder;
  }

  const result = await runOrchestrator(model, config, provider, budget, priorityOrder);
  result.stageA = stageA;
  return result;
}

function createProvider(config: AIConfig): AIProvider {
  if (config.provider === "mock") return new MockProvider({});
  if (config.provider === "anthropic") {
    if (!config.dryRun && !process.env.ANTHROPIC_API_KEY) {
      throw new Error("ANTHROPIC_API_KEY is not set. Run `creds set anthropic-api-key` or export the env var.");
    }
    // Lazy-load the SDK so tests and dry-run don't need it at import time.
    // eslint-disable-next-line @typescript-eslint/no-var-requires
    const Anthropic = require("@anthropic-ai/sdk").default;
    const client = new Anthropic({ apiKey: process.env.ANTHROPIC_API_KEY });
    return new AnthropicProvider(config.model, client);
  }
  if (config.provider === "openai") {
    if (!config.dryRun && !process.env.OPENAI_API_KEY) {
      throw new Error("OPENAI_API_KEY is not set. Run `creds set openai-api-key` or export the env var.");
    }
    // eslint-disable-next-line @typescript-eslint/no-var-requires
    const OpenAI = require("openai").default;
    const client = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });
    return new OpenAIProvider(config.model, client);
  }
  throw new Error(`Unknown AI provider: ${config.provider}`);
}

function pricingFor(config: AIConfig): ProviderPricing {
  if (config.provider === "anthropic") {
    return ANTHROPIC_PRICING[config.model] ?? ANTHROPIC_PRICING["claude-sonnet-4-6"];
  }
  if (config.provider === "openai") {
    return OPENAI_PRICING[config.model] ?? OPENAI_PRICING["gpt-5"];
  }
  return { inputPerMTokUsd: 0, outputPerMTokUsd: 0, cacheReadPerMTokUsd: 0, cacheWritePerMTokUsd: 0 };
}
```

```ts
function mergeAIConfig(config: SentinelConfig, options: ScanOptions): AIConfig {

export async function runScan(options: ScanOptions): Promise<number> {
  const config = await loadConfig(options.path);

  const reconStart = Date.now();
  const model = await runRecon(options.path, config.exclude);
  const recon_ms = Date.now() - reconStart;

  resetRegistry();
  registerRules(secretsExposureRules);
  registerRules(inputInjectionRules);
  registerRules(authAccessControlRules);
  registerRules(fileUploadRules);
  registerRules(supplyChainRules);
  registerRules(cryptographyRules);
  registerRules(loggingMonitoringRules);
  registerRules(apiSecurityRules);
  registerRules(businessLogicRules);
  registerRules(infrastructureRules);

  const rulesStart = Date.now();
  const ruleFindings = await runRules(model);
  const rules_ms = Date.now() - rulesStart;

  let aiResult: AIRunResult | null = null;
  let ai_ms = 0;
  const aiConfig = mergeAIConfig(config, options);

  if (aiConfig.enabled || options.aiDryRun) {
    const aiStart = Date.now();
    if (options.aiDryRun) {
      // Dry run path — emit the estimate and exit 0 without modifying findings.
      const dryConfig = { ...aiConfig, dryRun: true };
      const provider = dryConfig.provider === "mock" ? new MockProvider({}) : (await import("../layers/ai/index.js")).runAI;
      const estimate = await runDryRun(model, dryConfig, new MockProvider({}));
      process.stdout.write(formatDryRunReport(estimate, dryConfig) + "\n");
      return 0;
    }

    aiResult = await runAI(model, aiConfig);
    ai_ms = Date.now() - aiStart;
    model.findings.push(...aiResult.findings);
  }

  const allFindings = [...ruleFindings, ...(aiResult?.findings ?? [])];

  if (options.format === "json") {
    const body = formatJson(model, allFindings, { recon_ms, rules_ms, ai_ms, ai: aiResult ?? undefined });
    if (options.outFile) {
      const dest = resolve(options.outFile);
      await mkdir(dirname(dest), { recursive: true });
      await writeFile(dest, body, "utf-8");
      process.stdout.write(`Wrote JSON report to ${dest}\n`);
    } else {
      process.stdout.write(body);
    }
  } else if (options.format === "html") {
    const body = formatHtml(model, allFindings, { recon_ms, rules_ms });
    const dest = resolve(options.outFile ?? join(options.path, config.output.dir, `scan-${Date.now()}.html`));
    await mkdir(dirname(dest), { recursive: true });
    await writeFile(dest, body, "utf-8");
    process.stdout.write(`Wrote HTML report to ${dest}\n`);
  } else {
    process.stdout.write(formatTerminal(model, allFindings));
    if (aiResult) {
      process.stdout.write("\n" + formatAIReport(aiResult, aiConfig) + "\n");
    }
    process.stdout.write("\n");
  }

  const failOn = options.failOn ?? config.severity.failOn;
  const hasBlockingFinding = allFindings.some((f) => isAtOrAbove(f.severity, failOn));
  return hasBlockingFinding ? 1 : 0;
}

function formatAIReport(result: AIRunResult, config: AIConfig): string {
  const findingCounts = result.findings.reduce<Record<string, number>>((a, f) => {
    a[f.severity] = (a[f.severity] ?? 0) + 1;
    return a;
  }, {});
  const counts = ["critical", "high", "medium", "low"]
    .filter((s) => findingCounts[s])
    .map((s) => `${s}: ${findingCounts[s]}`)
    .join(", ");
  // NOTE: input_tokens is the FRESH (uncached) portion only; cache_read_input_tokens is a
  // separate disjoint bucket. The correct formula is cached / (fresh + cached).
  // Using cached / fresh (the old formula) can exceed 100% and is wrong.
  const totalInputConsumed = result.usage.inputTokens + result.usage.cacheReadTokens;
  const cacheRate =
    totalInputConsumed > 0
      ? `${Math.round((result.usage.cacheReadTokens / totalInputConsumed) * 100)}%`
      : "0%";
  return `AI Analysis
───────────
Endpoints analyzed: ${result.endpointsAnalyzed} / ${result.endpointsAnalyzed + result.endpointsSkipped}
Findings:           ${result.findings.length}${counts ? ` (${counts})` : ""}
Errors:             ${result.errors.length}
Tokens:
  Input:       ${result.usage.inputTokens.toLocaleString()}
  Cached read: ${result.usage.cacheReadTokens.toLocaleString()} (cache hit rate: ${cacheRate})
  Output:      ${result.usage.outputTokens.toLocaleString()}
Cost estimate: $${result.usage.estimatedCostUsd.toFixed(2)} (cap $${config.maxCostUsd.toFixed(2)})
Duration: ${Math.round(result.durationMs / 1000)}s${result.budgetExhausted ? "\nWARNING: budget exhausted — scan halted early" : ""}`;
}

──────────

Project summary tokens: ${e.projectSummaryTokens.toLocaleString()} (cached after first call)

Total input tokens: ~${e.totalInputTokens.toLocaleString()}
Estimated cost: $${e.estimatedCostUsd.toFixed(2)} (${config.provider} ${config.model})
Budget cap: $${e.maxCostUsd.toFixed(2)}
Status: ${e.withinBudget ? "within budget" : "EXCEEDS BUDGET"}

Run without --ai-dry-run to execute.`;
}
```
